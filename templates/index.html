Arquitetura Backend, SQL & Script Python Flask

-- SCRIPT SQL SCHEMA (MySQL / PostgreSQL)

CREATE DATABASE IF NOT EXISTS novo_horizonte_db;
USE novo_horizonte_db;

-- 1. Tabela de Unidades
CREATE TABLE unidades (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    endereco VARCHAR(255),
    ativo TINYINT(1) DEFAULT 1,
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Tabela de Usuários do Painel Admin
CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(150) NOT NULL,
    username VARCHAR(80) NOT NULL UNIQUE,
    senha_hash VARCHAR(255) NOT NULL,
    perfil VARCHAR(50) DEFAULT 'ADMIN',
    ativo TINYINT(1) DEFAULT 1
);

-- 3. Tabela de Solicitacoes
CREATE TABLE solicitacoes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    protocolo VARCHAR(50) NOT NULL UNIQUE,
    unidade_id INT NOT NULL,
    responsavel VARCHAR(150) NOT NULL,
    status ENUM('Pendente','Em análise','Aprovada','Em separação','Atendida','Recusada') DEFAULT 'Pendente',
    observacoes TEXT,
    observacao_admin TEXT,
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (unidade_id) REFERENCES unidades(id)
);

-- 4. Tabela de Itens da Solicitação
CREATE TABLE itens_solicitacao (
    id INT AUTO_INCREMENT PRIMARY KEY,
    solicitacao_id INT NOT NULL,
    nome_produto VARCHAR(150) NOT NULL,
    categoria VARCHAR(100) NOT NULL,
    quantidade DECIMAL(10,2) NOT NULL,
    unidade_medida VARCHAR(20) NOT NULL,
    observacao VARCHAR(255),
    FOREIGN KEY (solicitacao_id) REFERENCES solicitacoes(id) ON DELETE CASCADE
);

-- 5. Tabela de Histórico e Auditoria
CREATE TABLE historico_audit (
    id INT AUTO_INCREMENT PRIMARY KEY,
    solicitacao_id INT NOT NULL,
    usuario VARCHAR(100) NOT NULL,
    acao VARCHAR(255) NOT NULL,
    detalhes TEXT,
    data_hora TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

# ESTRUTURA SERVIDOR FLASK (Python REST API)

        return jsonify({'error': 'Campos obrigatórios ausentes'}), 400
    
    protocolo = f"#ANH-{datetime.datetime.now().year}-{int(datetime.datetime.now().timestamp())}"
